核心代码（简化版）

// 通用包装：把 setTimeout 变成 Promise
function an_delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
}

// 每个步骤都是异步函数，返回自己的名字 + 耗时
async function an_doStep(name, ms) {
    an_highlight(name, "active");
    await an_delay(ms);
    an_highlight(name, "done");
    return { name, ms };
}

// 主流程
async function an_run() {
    // 1️⃣ 2️⃣ 顺序等待
    await an_doStep("buy",    6000);   // 买原料 6s
    await an_doStep("gather", 2000);   // 取原料 2s

    // 3️⃣ 并行！和面和馅同时开始，取最长的那个完成
    await Promise.all([
        an_doStep("dough", 3000),      // 和面 3s
        an_doStep("fill",  4000),      // 做馅 4s
    ]);  // 实际用了 4s（不是 3+4=7s）

    // 4️⃣ 5️⃣ 继续顺序
    await an_doStep("wrap",  5000);    // 包饺子 5s
    await an_doStep("eat",   3000);    // 吃饺子 3s
}
